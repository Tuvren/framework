#!/bin/sh
set -eu

# Run from the EPIC-BZ repository checkout with Bun, direnv/devenv, and the
# locked workspace dependencies available. The raw captures are written to
# M19_REPLAY_DIR (default: ${TMPDIR:-/tmp}/epic-bz-m19-replay).
ROOT=$(git rev-parse --show-toplevel)
cd "$ROOT"

# Re-enter through direnv using POSIX sh, with one outer shared lock. This
# preserves the M19 wrapper environment without evaluating zsh output.
if [ "${1:-}" != --locked ]; then
  exec flock -x /tmp/tuvren-wave-1-verification.lock \
    direnv exec "$ROOT" env \
      NX_DAEMON=false \
      PGHOST=/run/user/1000/devenv-3ea1e50/postgres \
      PGPORT=5432 \
      sh "$ROOT/.constitution/evidence/KRT-BZ004/M19-replay-commands.txt" --locked
fi
shift

export NX_DAEMON=false
export PGHOST=/run/user/1000/devenv-3ea1e50/postgres
export PGPORT=5432
test -S "$PGHOST/.s.PGSQL.$PGPORT"

REPLAY_DIR=${M19_REPLAY_DIR:-${TMPDIR:-/tmp}/epic-bz-m19-replay}
SCRATCH=$(mktemp -d)
mkdir -p "$REPLAY_DIR"
cleanup() {
  git worktree remove --force "$SCRATCH/before" 2>/dev/null || true
  git worktree remove --force "$SCRATCH/after" 2>/dev/null || true
  rm -rf "$SCRATCH"
}
trap cleanup EXIT HUP INT TERM

# M19 source before/after revisions: 1213e15 is the parent; 8c03754 is the
# source change. Share the already-installed, lockfile-pinned dependencies.
git worktree add --quiet --detach "$SCRATCH/before" 1213e15bf6406763f5ef443d133c904c8078c6da
git worktree add --quiet --detach "$SCRATCH/after" 8c03754281b057b33058f852230a8ea392692fb8
ln -s "$ROOT/node_modules" "$SCRATCH/before/node_modules"
ln -s "$ROOT/node_modules" "$SCRATCH/after/node_modules"

capture_oxlint() {
  label=$1
  tree=$2
  expected_status=1
  shift 2
  if (cd "$SCRATCH/$tree" && bunx --bun oxlint "$@" \
      typescript/runners/react/src/lib/react-runner-around-model.ts) \
      >"$REPLAY_DIR/$label.json" 2>"$REPLAY_DIR/$label.stderr"; then
    status=0
  else
    status=$?
  fi
  if [ "$status" -ne "$expected_status" ]; then
    echo "$label: oxlint exit $status; expected $expected_status" >&2
    return 1
  fi
}

# Raw JSON captures use the unchanged 419-rule syntactic and 474-rule
# type-aware effective presets; status 1 is expected because findings exist.
capture_oxlint before-syntactic before --format=json
capture_oxlint before-typeaware before --type-aware --format=json
capture_oxlint after-syntactic after --format=json
capture_oxlint after-typeaware after --type-aware --format=json

# Exact-file formatting and retained-Biome checks from M19.
bunx --bun oxfmt --check \
  typescript/runners/react/src/lib/react-runner-around-model.ts
bunx --bun @biomejs/biome check \
  typescript/runners/react/src/lib/react-runner-around-model.ts

# Runner package gates.
bun ./tools/run-nx.mjs run runner-react:lint
bun ./tools/run-nx.mjs run runner-react:typecheck
bun ./tools/run-nx.mjs run runner-react:test
bun ./tools/run-nx.mjs run runner-react:bench
bun ./tools/run-nx.mjs run runner-react:build

# Repository and provider conformance gates.
bun run nx run-many -t typecheck --parallel=4
bun tools/scripts/conformance.ts
bun ./tools/run-nx.mjs run providers-testkit:conformance
bun ./tools/run-nx.mjs run providers-typescript-certification:conformance
bun tools/scripts/changeset-check.ts

# The constitution CLI is supplied by the installed skill; CODEX_HOME may be
# set by the host, with ~/.codex as the documented default.
CODEX_HOME=${CODEX_HOME:-$HOME/.codex}
bun run "$CODEX_HOME/skills/constitution-tooling/scripts/cli.ts" validate --root "$ROOT"

printf 'M19 replay captures: %s\n' "$REPLAY_DIR"
